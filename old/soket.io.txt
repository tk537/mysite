補足情報
このレッスンの内容は最新バージョンの Socket.IO では動作しません




2015年6月16日





Socket.IO の 1.0 から socket.set と socket.get メソッドが廃止されたため、そのまま動作させるとエラーが発生してしまいます。

エラーを回避するためには app.js で socket.set と socket.get を使用している箇所を以下のように修正します。
io.sockets.on('connection', function(socket) {
    socket.on('emit_from_client', function(data) {
        // ここから修正箇所
        socket.client_name = data.name; 
        io.sockets.emit('emit_from_server', '[' + socket.client_name + '] : ' + data.msg);
        // 修正箇所ここまで
    });
});

io.sockets.on('connection', function(socket) {
    socket.on('emit_from_client', function(data) {
        socket.set('client_name', data.name);
        socket.get('client_name', function(err, name) {
            io.sockets.emit('emit_from_server', '[' + name + '] : ' + data.msg);
        });
    });
});

var chat = io.of('/chat').on('connection', function(socket) {
    socket.on('emit_from_client', function(data) {
        socket.join(data.room);
        socket.emit('emit_from_server', 'you are in ' + data.room);
        socket.broadcast.to(data.room).emit('emit_from_server', data.msg);
    });
});
var news = io.of('/news').on('connection', function(socket) {
    socket.emit('emit_from_server', 'today: ' + new Date());
});
