@h1-color: red;
@h2-color: darken(@h1-color, 10%);

@h2-size: @h1-size - 4px;

@header: .myheader;
@color: color;
@dir: "/img";

@{header}1 {
    font-size: @h1-size;
    @{color}: @h1-color;
}

    font-size: @h2-size;

  font-size: 18px;
}

  color: red;
}
header a:hover {
  color: green;
}
.button-ok {
  color: green;
}
.button-ng {
  color: red;
}

// ===== styles.less =====
/*
header {
    font-size: 18px;
}

header a {
    color: red;
}
*/

header {
    font-size: 18px;
    a {
        color: red;
        &:hover {
            color: green;
        }
    }
}

.button {
    &-ok {
        color: green;
    }
    &-ng {
        color: red;
    }
}
