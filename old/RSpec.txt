// ===== commentBox 1 =====
補足情報
:context と :example は :all と :each のエイリアス




2014年6月2日





before に与えるオプションの :context や :example はそれぞれ :all と :each のエイリアスであり、RSpec 2 以前ではそれらが使用されています。

// ===== /lib/calc.rb =====
class Calc
  def add(a, b)
    # 5 # 仮実装
    a + b # 明らかな実装
  end
end

// ===== /spec/calc_spec.rb =====
require 'calc'

RSpec.describe "A calc" do 
  before do 
    @calc = Calc.new
  end
  it "given 2 and 3, returns 5" do
    expect(@calc.add(2, 3)).to eq(5)
  end
  # 三角測量
  it "given 5 and 8, returns 13" do
    expect(@calc.add(5, 8)).to eq(13)
  end
end

// ===== video_summary =====
letについて気をつけるべき点を説明したのちに、let!との違いについても触れていきます。


    letの注意点
    let!との違い

// ===== /lib/calc.rb =====
class Calc
  attr_accessor :tax
  def initialize
    @tax = 0
  end
  def price(x) 
    x * (1 + tax)
  end
end

// ===== /spec/calc_spec.rb =====
require 'calc'

RSpec.describe Calc do 
  subject(:calc) { Calc.new }
  # let
  # 1. exampleごとに結果がキャッシュされる
  # 2. 遅延評価
  let!(:tax) { 
    calc.tax = 0.05
  }
  it { expect(tax).to eq(0.05) }
  it { expect(calc.price(100)).to eq(105) }
end

// ===== /lib/calc.rb =====
class Calc

  def add(a, b, name)
    (a + b).to_s + ' by ' + name
  end

end

// ===== /spec/calc_spec.rb =====
require 'calc'

# test double: 代役オブジェクト
# method stub

RSpec.describe Calc do 
  it { 
    user = double('user')
    allow(user).to receive(:name).and_return('taguchi')
    # user.name -> taguchi
    calc = Calc.new
    expect(calc.add(5, 2, user.name)).to eq('7 by taguchi')
  }
end

    @logger = logger
  end

    @logger.log
    a + b
  end

end

# test double: 代役オブジェクト
# message expectation: 呼ばれなかったらfail

  it { 
    logger = double('logger')
    expect(logger).to receive(:log)
    calc = Calc.new(logger)
    expect(calc.add(5, 2)).to eq(7)
  }
