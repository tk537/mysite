def get_price(a, b):
    # 仮引数、値が代入された変数 → ローカル変数
    # 値が代入されていない変数 → 外のスコープを探す
    global rate
    if a + b >= 3000:

    total = (a + b) * rate
    return total

rate = 1.1

globalキーワードによる問題を、値を引数で渡すことで解決していきます。

値の引数渡し



main.py
def get_price(a, b, rate):
    # 仮引数、値が代入された変数 → ローカル変数
    # 値が代入されていない変数 → 外のスコープを探す
    # global rate
    if a + b >= 3000:
        rate = 1.05
    total = (a + b) * rate
    return total

rate = 1.1
print(get_price(300, 700, rate))
print(get_price(3000, 7000, rate))
print(rate)
