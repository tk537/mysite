/src/App.vue 
<script setup>
import { ref, reactive } from 'vue';

const todos = reactive([
  {
    id: crypto.randomUUID(),
    title: 'Item 0',
  },
  {
    id: crypto.randomUUID(),
    title: 'Item 1',
  },
  {
    id: crypto.randomUUID(),
    title: 'Item 2',
  },
]);

const newTodo = ref('');

function addTodo() {
  if (newTodo.value.trim() === '') {
    return;
  }

  todos.push({
    id: crypto.randomUUID(),
    title: newTodo.value,
  });
  newTodo.value = '';
}
</script>

<template>
  <div class="container">
    <h1>Todos</h1>
    <ul>
      <li v-for="todo in todos" :key="todo.id">{{ todo.title }}</li>
    </ul>
    <form @submit.prevent="addTodo">
      <input type="text" v-model="newTodo">
      <button>Add</button>
    </form>
  </div>
</template>
