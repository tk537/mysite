// ===== video_summary =====
条件付きでドキュメントを抽出する方法について見ていきます。


    ドキュメントの抽出
    フィールドに含まれる値を抽出

// ===== read.me =====
// テストデータ作成用コマンドはこちらをご利用ください
db.users.insert({name: "taguchi", score: 52, team: "team-1"});
db.users.insert({name: "fkoji", score: 82, team: "team-2"});
db.users.insert({name: "dotinstall", score: 66, team: "team-3"});
db.users.insert({name: "yamada", score: 26, team: "team-1"});
db.users.insert({name: "kimura", score: 29, team: "team-2"});

db.users.find();

// team が team-1 のドキュメントを抽出
db.users.find({team: "team-1"});

// score が 50 以上のドキュメントを抽出
db.users.find({score: {$gte: 50}});

// $gte $gt $lte $lt
// $eq $ne

// team が team-1 のドキュメントを抽出
db.users.find({team: {$eq: "team-1"}});

// name に t を含むドキュメントを抽出
db.users.find({name: /t/});

// name が t で始まるドキュメントを抽出
db.users.find({name: /^t/});

// team に存在する値を抽出
db.users.distinct("team");


// ===== video_summary =====
より複雑な条件でドキュメントを抽出する方法をみていきます。


    複数の条件による抽出
    $or
    $in
    $exists

// ===== read.me =====
// 名前に i を含む、なおかつ、score が 50 以上のドキュメントを抽出
db.users.find({name:/i/, score:{$gte:50}});

// 名前に i を含む、もしくは、score が 50 以上のドキュメントを抽出
db.users.find({$or: [{name:/i/}, {score:{$gte:50}}]});

// score が 52、もしくは 66 のドキュメントを抽出
db.users.find({score: {$in: [52, 66]}});

// データ構造の異なるドキュメントを挿入
db.users.insert({name:"tanaka", score:52, age:23});

// age フィールドが存在するドキュメントを抽出
db.users.find({age: {$exists: true}});

// ===== read.me =====

db.users.find({}, {name: true, score: 1});

// 全件抽出だが、score を非表示
db.users.find({}, {score: 0});

// エラー
db.users.find({}, {score: 0, name: 1});

// score は表示するが、_id は表示しない
db.users.find({}, {score: 1, _id: 0});


// ===== video_summary =====
抽出結果を変更するためのコマンドについて見ていきます。


    sort()
    limit()
    findOne()
    skip()

// ===== read.me =====
// score が小さい順に抽出
db.users.find({}, {_id: 0}).sort({score: 1});

// score が大きい順に抽出
db.users.find({}, {_id: 0}).sort({score: -1});

// 3 件のみ抽出
db.users.find({}, {_id: 0}).limit(3);

// score が高い 3 件のみ抽出
db.users.find({}, {_id: 0}).sort({score: -1}).limit(3);

// 1 件のみ抽出
db.users.findOne({}, {_id: 0});

// 全件抽出
db.users.find({}, {_id: 0});

// 2 件目から表示（0 から開始）
db.users.find({}, {_id: 0}).skip(2);


// ===== video_summary =====
ドキュメントを更新するためのコマンドについて見ていきます。


    update()
    multi

// ===== read.me =====
db.users.find({name:"taguchi"}, {_id:0});

// score を 80 に更新
db.users.update({name: "taguchi"}, {$set: {score: 80}});

db.users.find({name:"taguchi"}, {_id:0});

// score を 90、team を team-2 に更新
db.users.update({name: "taguchi"}, {$set: {score: 90 ,team: "team-2"}});

db.users.find({name:"taguchi"}, {_id:0});

// name が taguchi のドキュメントを、name が taguchi、score を 40 に置き換え
db.users.update({name: "taguchi"}, {name: "taguchi", score: 40});

db.users.find({name:"taguchi"}, {_id:0});

db.users.find();

// team が team-2 のドキュメントの score を 100 に更新（最初の1件のみ）
db.users.update({team:"team-2"}, {$set: {score: 100}});

db.users.find();

// team が team-2 のドキュメントの score を 0 に更新（すべて）
db.users.update({team:"team-2"}, {$set: {score: 0}}, {multi: true});


// ===== video_summary =====
ドキュメントを変更するためのさまざまなコマンドについて見ていきます。


    $inc
    $mul
    $rename
    $set
    $unset

// ===== read.me =====
db.users.find({name: "taguchi"}, {_id:0});

// score に 5 を加える
db.users.update({name: "taguchi"}, {$inc: {score: 5}});

db.users.find({name: "taguchi"}, {_id:0});

// score を 2 倍する
db.users.update({name: "taguchi"}, {$mul: {score: 2}});

db.users.find({name: "taguchi"}, {_id:0});

// score を point に変更
db.users.update({name: "taguchi"}, {$rename: {score: "point"}});

db.users.find({name: "taguchi"}, {_id:0});

// team を追加
db.users.update({name: "taguchi"}, {$set: {team: "team-4"}});

db.users.find({name: "taguchi"}, {_id:0});

// team を削除
db.users.update({name: "taguchi"}, {$unset: {team: ""}});

db.users.find({name: "taguchi"}, {_id:0});


// ===== video_summary =====
ドキュメントが存在していたら更新、そうでなければ新規作成してくれるupsert、それから条件付きでドキュメントを削除する方法について見ていきます。


    upsert
    remove()

// ===== read.me =====
db.users.find();

// name が kato のドキュメントを更新（変更なし）
db.users.update({name: "kato"}, {name: "kato", score: 48});

db.users.find();

// name が kato のドキュメントを作成
db.users.update({name: "kato"}, {name: "kato", score: 48}, {upsert: true});

db.users.find();

// name が kato のドキュメントを更新
db.users.update({name: "kato"}, {name: "kato", score: 99}, {upsert: true});

db.users.find();

db.users.remove({name: "kato"});

db.users.find();
