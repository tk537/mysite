// ===== commentBox 1 =====
補足情報
【訂正】 default 'now'について




2014年4月3日





今回ご紹介したコード中、以下の行に誤りがありましたので訂正いたします。
created timestamp default 'now'

この設定では「レコードを挿入した日時」とはならず、「テーブルを作成した日時」になってしまいます。
createdフイールドを「レコードを挿入した日時」にしたい場合は以下のように修正してください。
created timestamp default statement_timestamp()

// ===== read.me =====
create table posts (
    id serial primary key,
    title varchar(255) not null, 
    body text check(length(body) > 5),
    is_draft boolean default TRUE,
    created timestamp default 'now'
);
-- コメント
/*

真偽: boolean TRUE FALSE t f

補足情報

1:39あたりで説明されている「@」ですが、そのあとで例を出して説明しているとおり、「%」の間違いでした...。任意の文字を表す記号には「%」を使うようにしてください。

-- > < <= >= = <> !=
select * from users where score = 5.0;
select * from users where score != 5.0;
-- like @ _

select * from users where name like '%i';
select * from users where name like 'sa_aki';

// ===== video_summary =====
複数の命令を確実にまとめて処理するためのトランザクションについて見ていきます。


    トランザクションの概要
    begin; commit;
    begin; rollback;

// ===== read.me =====
select * from users;
begin;
update users set score = score - 1.0 where name = 'taguchi';
update users set score = score + 1.0 where name = 'fkoji';
commit;
begin;
update users set score = score - 1.0 where name = 'taguchi';
update users set score = score + 1.0 where name = 'fkoji';
select * from users;
rollback;
select * from users;
