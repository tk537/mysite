'use strict';

{
  let t = 0;
  
  function draw() {
    const canvas = document.querySelector('canvas');
    if (typeof canvas.getContext === 'undefined') {
      return;
    }
    const ctx = canvas.getContext('2d');
    
    ctx.clearRect(0, 0, canvas.width, canvas.height);
    
    ctx.beginPath();
    ctx.ellipse(100, 100, 40, 80, 0, 0, 2 * Math.PI);
    ctx.fillStyle = 'black';
    ctx.fill();
    
    ctx.beginPath();
    ctx.ellipse( 80 + Math.sin(t / 30), 100, 8, 8, 0, 0, 2 * Math.PI);
    ctx.ellipse(120 + Math.sin(t / 30), 100, 8, 8, 0, 0, 2 * Math.PI);
    ctx.fillStyle = 'skyblue';
    ctx.fill();
    
    t++;
    setTimeout(draw, 10);
 }

  draw();
}

shadowBlurの単位について




2021年7月31日





0:55〜からshadowBlurを4「px」と表現しておりますが、単位はありません。shadowBlurはぼかし具合を示す正の値を指定します。より詳しくは以下もご確認ください。

https://developer.mozilla.org/en-US/docs/Web/API/CanvasRenderingContext2D/shadowBlur

    ctx.shadowOffsetY = 4;
    ctx.shadowBlur = 4;

{
  function draw() {
    const canvas = document.querySelector('canvas');
    if (typeof canvas.getContext === 'undefined') {
      return;
    }
    const ctx = canvas.getContext('2d');

    ctx.beginPath();
    ctx.moveTo(0, 100);
    ctx.lineTo(canvas.width, 100);
    ctx.moveTo(100, 0);
    ctx.lineTo(100, canvas.height);
    ctx.stroke();

    ctx.font = 'bold 64px Verdana';
    ctx.textAlign = 'right';
    ctx.textBaseline = 'top';

    // ctx.fillText('Tokyo', 100, 100);
    // ctx.fillText('Tokyo', 100, 100, 100);

高解像度ディスプレイか調べる方法

今回の内容は、高解像度ディスプレイをお使いの場合にだけ有効です。使っているディスプレイが高解像度かどうかを確認するには、
const dpr = window.devicePixelRatio || 1;
のあとで、
console.log(dpr);
としてみてください。こちらで「1」と表示された場合は高解像度ディスプレイに対応していない、ということになります。

// ===== /js/main.js =====
'use strict';

{
  let t = 0;

  function draw() {
    const canvas = document.querySelector('canvas');
    if (typeof canvas.getContext === 'undefined') {
      return;
    }
    const ctx = canvas.getContext('2d');

    const CANVAS_WIDTH = 600;
    const CANVAS_HEIGHT = 240;
    const dpr = window.devicePixelRatio || 1;
    canvas.width = CANVAS_WIDTH * dpr;
    canvas.height = CANVAS_HEIGHT * dpr;
    ctx.scale(dpr, dpr);
    canvas.style.width = CANVAS_WIDTH + 'px';
    canvas.style.height = CANVAS_HEIGHT + 'px';

    ctx.font = 'bold 48px Verdana';
    ctx.strokeText('Tokyo', 100, 100);
  }

  draw();
}
