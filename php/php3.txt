// ===== main.php =====
<?php
function getTotal($price, $amount, $rate = 1.1)
{
    if ($amount >= 100) {
        return $price * $amount;
    } else {
        return $price * $amount * $rate;
    }
}

// ===== main.php =====
<?php
declare(strict_types=1);

// function getProductCode(int $id): null|string
function getProductCode(int $id): ?string
{
    if ($id >= 100) {
        return NULL;
    }
    return "P-{$id}";
}
