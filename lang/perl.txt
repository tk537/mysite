    引数

// ===== hello.pl =====
use strict;
use warnings;

# サブルーチン

sub max {
    my $max = $_[0];
    if ($_[1] > $max) {
        $max = $_[1];
    }
    $max;
}

print max(12, 8);





// ===== video_summary =====
前回作ったサブルーチンをさらに改良していきます。


    引数の取得
    ロジックの改良

// ===== hello.pl =====
use strict;
use warnings;

# サブルーチン

sub max {
    my $max = shift; # @_
    for (@_) {
        $max = $_ if $_ > $max;
    }
    $max;
}

print max(12, 8, 100, 200, 300, 23, 38);
