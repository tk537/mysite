// ===== commentBox 1 =====
補足情報
Bluetooth のスピーカー等に接続していると起動エラーになる場合があります




2018年4月20日





ユーザーの方よりご報告いただきましたが、Bluetooth のスピーカーやヘッドホンに接続したまま Sonic PI を起動するとエラーになる場合があります。

その場合は接続を解除してから起動してください。

また、起動後にスピーカーやヘッドホンに接続し直しても、音はパソコンのほうから出力される可能性があります。

// ===== commentBox 2 =====
補足情報
「開発元が未確認のため開けません」と出る場合




2018年4月19日





macOSでインストールする際に「開発元が未確認のため開けません」と出る場合、アプリケーションフォルダ内の「Sonic Pi」のアイコンで右クリックしてから「開く」を選択してください。

// ===== house.rb =====
use_bpm 128

live_loop :kick do
  with_fx :compressor do
    sample :bd_sone, amp: 3 if (ring 1, 0, 1, 0, 1, 0, 1, 0).tick == 1
    sleep 0.5
  end
end

live_loop :sd do
  sleep 1
  sample :sn_dolf, amp: 1
  sleep 1
end

live_loop :hh do
  with_fx :echo, mix: 0.2, decay: 5 do
    sleep 0.5
    sample :drum_cymbal_closed, amp: 0.7
    sleep 0.5
  end
end

live_loop :piano do
  with_fx :lpf, cutoff: 120 do
    with_synth :piano do
      play chord(:d4, 'm7+9'), decay: 2, sustain: 2, relese: 2, amp: 3
      sleep 4
      play chord(:g3, '7-9'), decay: 2, sustain: 2, relese: 4, amp: 3
      sleep 4
      play chord(:c4, :maj9), decay: 2, sustain: 2, relese: 4, amp: 3
      sleep 4
      play chord(:f4, :maj9), decay: 2, sustain: 2, relese: 4, amp: 3
      sleep 4
      play chord(:b3, 'm7-5'), decay: 2, sustain: 2, relese: 4, amp: 3
      sleep 4
      play chord(:e3, '7+5-9'), decay: 2, sustain: 2, relese: 4, amp: 3
      sleep 4
      play chord(:a3, :m9), decay: 2, sustain: 2, relese: 4, amp: 3
      sleep 4
      play chord(:a3, '7+5-9'), decay: 2, sustain: 2, relese: 4, amp: 3
      play :Db3
      sleep 4
    end
  end
end

live_loop :pianom do
  use_random_seed 10
  with_fx :lpf, cutoff: 100 do
    with_fx :reverb, mix: 0.6 do
      with_synth :piano do
        play_pattern_timed chord(:d4, 'm7+9').shuffle, 0.5
        sleep 2
        play_pattern_timed chord(:g3, '7-9').shuffle, 0.5
        sleep 2
        play_pattern_timed chord(:c4, :maj9).shuffle, 0.5
        sleep 1.5
        play_pattern_timed chord(:f4, :maj9).shuffle, 0.5
        sleep 2
        play_pattern_timed chord(:b3, 'm7-5').shuffle, 0.5
        sleep 1.5
        play_pattern_timed chord(:e3, '7+5-9').shuffle, 0.5
        sleep 1
        play_pattern_timed chord(:a3, :m9).shuffle, 0.5
        sleep 0.5
        play_pattern_timed chord(:a3, '7+5-9').shuffle, 0.5
        play :Db3
        sleep 2
      end
    end
  end
end

live_loop :bass do
  with_fx :tremolo, amp: 3 do
    with_synth :piano do
      play_pattern_timed [:d2, :g1, :c2, :f2, :b1, :e1, :a1, :Db2], 4
    end
  end
end
