① シンプルな再利用LLM関数（PromptTemplate + Runnable）
# llm_runnable_example1.py
from langchain_core.prompts import PromptTemplate
from langchain_core.runnables import Runnable
from openai import OpenAI

api_key = "YOUR_API_KEY"
llm_client = OpenAI(api_key=api_key)

# PromptTemplateでテンプレート化
prompt_template = PromptTemplate(
    input_variables=["animal"],
    template="日本の{animal}について簡単に教えてください。"
)

class LLMRunnable(Runnable):
    def invoke(self, input_str: str) -> str:
        try:
            prompt_text = prompt_template.format(animal=input_str)
            response = llm_client.completions.create(
                model="gpt-4o-mini",
                prompt=prompt_text,
                max_tokens=150
            )
            return response.choices[0].text.strip()
        except Exception as e:
            return f"エラー発生: {e}"

# 再利用例
animal_chain = LLMRunnable()
print(animal_chain.invoke("猫"))
print(animal_chain.invoke("犬"))


✅ 特徴

PromptTemplateでテンプレート化

Runnableで再利用可能に

try/exceptで簡単なエラーハンドリング








② バッチ処理対応例
# llm_runnable_batch.py
from langchain_core.prompts import PromptTemplate
from langchain_core.runnables import Runnable
from openai import OpenAI

api_key = "YOUR_API_KEY"
llm_client = OpenAI(api_key=api_key)

prompt_template = PromptTemplate(
    input_variables=["animal"],
    template="日本の{animal}について教えてください。"
)

class LLMBatchRunnable(Runnable):
    def invoke(self, animals: list[str]) -> list[str]:
        results = []
        for animal in animals:
            try:
                prompt_text = prompt_template.format(animal=animal)
                response = llm_client.completions.create(
                    model="gpt-4o-mini",
                    prompt=prompt_text,
                    max_tokens=150
                )
                results.append(response.choices[0].text.strip())
            except Exception as e:
                results.append(f"エラー: {e}")
        return results

batch_chain = LLMBatchRunnable()
animals = ["猫", "犬", "パンダ"]
outputs = batch_chain.invoke(animals)
for o in outputs:
    print(o)


✅ 特徴

バッチ入力に対応

エラーがあっても途中で止まらず、個別結果として返す
