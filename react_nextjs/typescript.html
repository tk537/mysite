データ型をパラメータ化することができるジェネリクスについて見ていきます。

<>
ジェネリクス

'use strict';

{
  function printNumberTwice(num: number): void {
    console.log(num);
    console.log(num);
  }

  function printStringTwice(str: string): void {
    console.log(str);
    console.log(str);
  }

  function printTwice(value: T): void {
    console.log(value);
    console.log(value);
  }

  printTwice(10);
  printTwice('OK');
}
