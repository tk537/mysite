package main

import "fmt"

func main() {
    var score int
    fmt.Println("Score?")
    fmt.Scanf("%d", &score)

    if score >= 90 {
        fmt.Println("A!")
    } else {
        fmt.Println("Not A!")
    }
}

switchを使って条件分岐をする方法について見ていきます。

switch {case score >= 90: ... case score >= 70: ... default: ... }
switchの利点、見やすい
switchかifか、ケースバイケース

main.go
package main

import "fmt"

func main() {
    var score int
    fmt.Println("Score?")
    fmt.Scanf("%d", &score)

    // if score >= 90 {
    //     fmt.Println("A!")
    // } else if score >= 70 {
    //     fmt.Println("B!")
    // } else {
    //     fmt.Println("C!")
    // }

    switch {
    case score >= 90:
        fmt.Println("A!")
    case score >= 70:
        fmt.Println("B!")
    default:
        fmt.Println("C!")
    }
}






switchを使った別の条件分岐の方法について見ていきます。

scoreとある値が等しいかどうかで条件分岐
switch score { case 0: ... case 1: ... default: ... }
複数の場合、case 3, 4, 5: ...

main.go
package main

import "fmt"

func main() {
    var score int
    fmt.Println("Score?")
    fmt.Scanf("%d", &score)

    switch score {
    case 0:
        fmt.Println("Fail!")
    case 1:
        fmt.Println("C!")
    case 2:
        fmt.Println("B!")
    case 3, 4, 5:
        fmt.Println("A!")
    default:
        fmt.Println("Invalid score!")
    }
}

ループを途中で終了させたり、以降の処理をスキップして次の反復処理に移るための命令について見ていきます。

300円を超えた場合、処理を終了、break
200円以下のときは表示をスキップ、continue

main.go
package main

import "fmt"

func main() {
    money := 100.0

    for year := 1; year <= 30; year++ {
        money *= 1.05
        if money > 300 {
            break
        }
        if money < 200 {
            continue
        }
        fmt.Printf("Year %d: %f\n", year, money)
    }
}

より複雑な条件を作ることができる論理演算子について見ていきます。

A && B、AなおかつB
A || B、AまたはB
!A、Aではない
論理演算子を使った複雑な条件分岐
if english >= 80 && math >= 90 { ... }

main.go
package main

import "fmt"

func main() {
    var english int
    fmt.Println("English?")
    fmt.Scanf("%d", &english)
    var math int
    fmt.Println("Math?")
    fmt.Scanf("%d", &math)

    if english >= 80 && math >= 90 {
    if english >= 80 || math >= 90 {
    if !(english >= 80 && math >= 90) {
        fmt.Println("A")
    } else {
        fmt.Println("B")
    }
}
