const params = {
  width: 100,
  height: 100,
  depth: 100,
  backgroundColor: '#f00',
  canRotate: true,
};

  createCanvas(400, 200, WEBGL);

  });
  sizeFolder.addInput(params, 'width', {

    max: 200,

  width: 100,
  height: 100,
  depth: 100,

  canRotate: true,
};

function setup() {
  createCanvas(400, 200, WEBGL);

  const pane = new Tweakpane({
    title: 'Parameters',
  });
  const sizeFolder = pane.addFolder({
    title: 'Size',
  });
  sizeFolder.addInput(params, 'width', {
    min: 0,
    max: 200,
  });
  sizeFolder.addButton({
    title: 'Randomize width',
  }).on('click', () => {
    params.width = Math.random() * 100;
    pane.refresh();
  });
  sizeFolder.addInput(params, 'height', {

  });

      small: 50,

  pane.addSeparator();
