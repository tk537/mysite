// ===== video_summary =====
データの量が増えても対応できるように、domainとrangeを書き換えたのちに、グラフに余白をつけていきます。


    map()
    bandwidth()
    padding()

// ===== /MyD3App/index.html =====
<!DOCTYPE html>
<html lang="ja">
<head>
  <meta charset="UTF-8">
  <title>D3.js Practice</title>
</head>
<body>
  <svg width="600" height="240" style="background:#f8f8f8;">
  </svg>

  <script src="https://d3js.org/d3.v5.min.js"></script>
  <script>
    (function() {
      'use strict';

      var WIDTH = 600;
      var HEIGHT = 240;

      var scores = [
        {name: 'taguchi', score: 77},
        {name: 'fkoji', score: 92},
        {name: 'dotinstall', score: 68},
        {name: 'tanaka', score: 42},
        {name: 'yoshida', score: 88}
      ];

      function update(data) {
        var bars = d3.select('svg').selectAll('rect').data(data);

        var xScale = d3.scaleLinear()
          .domain([0, d3.max(scores, function(d) {
            return d.score;
          })])
          .range([0, WIDTH])
          .nice();

        var yScale = d3.scaleBand()
          .domain(scores.map(function(d) {
            return d.name;
          }))
          .range([0, HEIGHT])
          .padding([0.1]);

        bars
          .enter()
          .append('rect')
          .merge(bars)
          .attr('x', 0)
          .attr('y', function(d, i) {
            return yScale(d.name);
          })
          .attr('width', function(d) {
            return xScale(d.score);
          })
          .attr('height', yScale.bandwidth())
          .attr('fill', 'pink');

        bars
          .exit()
          .remove();
      }

      update(scores);
    })();
  </script>
</body>
</html>

    axisLeft()
    余白の設定

// ===== /MyD3App/index.html =====
<!DOCTYPE html>
<html lang="ja">
<head>
  <meta charset="UTF-8">
  <title>D3.js Practice</title>
</head>
<body>
  <svg width="600" height="240" style="background:#f8f8f8;">
  </svg>

  <script src="https://d3js.org/d3.v5.min.js"></script>
  <script>
    (function() {
      'use strict';

      var WIDTH = 600;
      var HEIGHT = 240;

      var padding = {
        top: 40,
        bottom: 40,
        left: 100,
        right: 40
      };

      var scores = [
        {name: 'taguchi', score: 77},
        {name: 'fkoji', score: 92},
        {name: 'dotinstall', score: 68},
        {name: 'tanaka', score: 42},
        {name: 'yoshida', score: 88}
      ];

      function update(data) {
        var bars = d3.select('svg').selectAll('rect').data(data);

        var xScale = d3.scaleLinear()
          .domain([0, d3.max(scores, function(d) {
            return d.score;
          })])
          .range([0, WIDTH - padding.left - padding.right])
          .nice();

        var yScale = d3.scaleBand()
          .domain(scores.map(function(d) {
            return d.name;
          }))
          .range([0, HEIGHT - padding.top - padding.bottom])
          .padding([0.1]);

        bars
          .enter()
          .append('rect')
          .merge(bars)
          .attr('x', padding.left)
          .attr('y', function(d, i) {
            return yScale(d.name) + padding.top;
          })
          .attr('width', function(d) {
            return xScale(d.score);
          })
          .attr('height', yScale.bandwidth())
          .attr('fill', 'pink');

        bars
          .exit()
          .remove();

        d3.select('svg')
          .append('g')
          .call(d3.axisBottom(xScale));

        d3.select('svg')
          .append('g')
          .call(d3.axisLeft(yScale));
      }

      update(scores);
    })();
  </script>
</body>
</html>

// ===== /MyD3App/index.html =====
<!DOCTYPE html>
<html lang="ja">
<head>
  <meta charset="UTF-8">
  <title>D3.js Practice</title>
</head>
<body>
  <svg width="600" height="240" style="background:#f8f8f8;">
  </svg>

  <script src="https://d3js.org/d3.v5.min.js"></script>
  <script>
    (function() {
      'use strict';

      var WIDTH = 600;
      var HEIGHT = 240;

      var padding = {
        top: 40,
        bottom: 40,
        left: 100,
        right: 40
      };

      var scores = [
        {name: 'taguchi', score: 77},
        {name: 'fkoji', score: 92},
        {name: 'dotinstall', score: 68},
        {name: 'tanaka', score: 42},
        {name: 'yoshida', score: 88}
      ];

      function update(data) {
        var bars = d3.select('svg').selectAll('rect').data(data);

        var xScale = d3.scaleLinear()
          .domain([0, d3.max(scores, function(d) {
            return d.score;
          })])
          .range([0, WIDTH - padding.left - padding.right])
          .nice();

        var yScale = d3.scaleBand()
          .domain(scores.map(function(d) {
            return d.name;
          }))
          .range([0, HEIGHT - padding.top - padding.bottom])
          .padding([0.1]);

        var color = d3.scaleOrdinal(d3.schemeCategory10);

        bars
          .enter()
          .append('rect')
          .merge(bars)
          .attr('x', padding.left)
          .attr('y', function(d, i) {
            return yScale(d.name) + padding.top;
          })
          .attr('width', 0)
          .attr('height', yScale.bandwidth())
          .attr('fill', function(d, i) {
            // return 'hsla(' + i * 30 + ', 60%, 60%, 0.9)';
            return color(i);
          })
          .transition()
            .duration(200)
            .delay(function(d, i) {
              return i * 200;
            })
            .ease(d3.easeElastic)
            .attr('width', function(d) {
              return xScale(d.score);

          .exit()
          .remove();
