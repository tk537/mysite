ControlP5の使い方を説明しつつライブラリの導入方法について見ていきます。


    ControlP5のインストール
    ControlP5の利用

// ===== sketch_141202c.pde =====
import controlP5.*;
ControlP5 cp5;

void setup() {
  size(200, 200);
  smooth();
  background(255);
  noStroke();
  fill(#ff0000);
  cp5 = new ControlP5(this);
  cp5.addSlider("r")
     .setPosition(10, 10)
     .setSize(100, 15)
     .setRange(0, 255);
}

int r = 100;

void draw() {
  background(255);
  ellipse(mouseX, mouseY, r, r);
}

void mousePressed() {
  r += 5;
}
