// ===== commentBox 1 =====
補足情報
cos, sinについて




2019年2月12日





動画中で

x = sin(radians(angle)) * r;
y = cos(radians(angle)) * r;


としていましたが、正しくは、x方向がcos、y方向がsinなので、

x = cos(radians(angle)) * r;
y = sin(radians(angle)) * r;


となります。

var angle = 0;
var r = 30;

function setup() {
  createCanvas(480, 240);
  background('skyblue');
  noStroke();
}

function draw() {
  push();
  translate(width/2, height/2);
  x = sin(radians(angle)) * r;
  y = cos(radians(angle)) * r;
  ellipse(x, y, 10, 10);
  pop();
  angle += 2;
  r += 0.1;
}

/*
p5.dom
*/

var angle = 0;
var r = 30;

var slider;
var button;

function setup() {
  createCanvas(480, 240);
  background('skyblue');
  noStroke();

  slider = createSlider(0, 100, 30);
  slider.position(10, 20);

  button = createButton("clear!");
  button.position(10, 40);
  button.mousePressed(function() {
    background('skyblue');
  });
}

function draw() {
  r = slider.value();
  push();
  translate(width/2, height/2);
  x = sin(radians(angle)) * r;
  y = cos(radians(angle)) * r;
  ellipse(x, y, 10, 10);
  pop();
  angle += 2;
  r += 0.1;
}
