class Post:
    _count = 0
    
    def __init__(self, text):
        self._text = text
        self._likes = 0
        Post._count += 1
    
    @classmethod
    def show_count(cls):
        print(f"{cls._count} instances created")
    
    def show(self):
        print(f"{self._text} - {self._likes}")
    
    def like(self):
        self._likes += 1

posts = [
    Post("Hello"),
    Post("Hi"),
]

for post in posts:
    post.show()

class Post: # 親クラス Superクラス
    def __init__(self, text):

    def like(self):

class SponsoredPost(Post): # 子クラス Subクラス
    def __init__(self, text, sponsor):
        # self._text = text

        super().__init__(text)

    def show(self):
        print(f"{self._text} - {self._likes} sponsored by {self._sponsor}")

補足情報 メソッドの種類について
2022年6月8日
クラス内では以下のような種類のメソッドを定義することができます。

インスタンスメソッド → インスタンスのデータに紐づくメソッド。第一引数に暗黙的にインスタンス自身（self）が渡される。
クラスメソッド → クラスのデータに紐づくメソッド。第一引数に暗黙的にクラス自身（cls）が渡される。@classmethodをつける。
静的（static）メソッド → インスタンスやクラスのデータに紐づいていないメソッド。第一引数に暗黙的に値を渡さないために@staticmethodをつける。

main.py

    @staticmethod

    @staticmethod
