より複雑なデータを集計して辞書にまとめる方法について見ていきます。

タプルの配列

main.py
from collections import defaultdict
results = [
    ("pass", "Taro"),
    ("pass", "Jiro"),
    ("fail", "Saburo"),
    ("pass", "Shiro")]
# stats = {"pass": ["Taro", "Jiro", "Shiro"], "fail": ["Saburo"]}

stats = defaultdict(list)

for result, name in results:
    stats[result].append(name)

print(dict(stats))
