マップとスライスを組み合わせて、少し複雑なデータについて見ていきます。

main.go
package main

import "fmt"

func main() {
    scores := map[string][]int{
        "english": []int{80, 40},
        "math":    []int{40, 50, 60},
    }

    for subject, scoresSlice := range scores {
        fmt.Println(subject)
        for _, score := range scoresSlice {
            fmt.Println(score)
        }
    }
}
