値を更新して再代入する際の表現について学びます。

値の再代入
+= -=
++ --

main.go
package main

import "fmt"

func main() {
    price := 150
    fmt.Println(price * 120)
    fmt.Println(price * 130)
    fmt.Println(price * 140)

    // price = 151
    // price = price + 1
    // price += 1
    price++
    // price -= 1
    // price--
    fmt.Println(price * 120)
    fmt.Println(price * 130)
    fmt.Println(price * 140)
}

補足情報 大文字からはじまる名前について
2024年11月6日
Go言語では変数や定数、そしてあとから見ていく関数やメソッド、構造体などの名前は、 他のファイル（正確にいうと他のpackage）からも使えるようにしたい場合、最初を大文字にする、というルールがあります。

たとえばPrintln()は大文字から始まっていますが、これはfmtのpackageで定義されていて、fmt以外でも使えるようにしたいため、先頭が大文字になっている、と理解しておきましょう。

main.go
package main

import "fmt"

func main() {
    // var fname string
    // fname = "Taro"
    fname := "Taro"
    lname := "Yamada"
    // fmt.Println("I am Taro Yamada")
    // fmt.Println("I am " + fname + " " + lname)
    fmt.Printf("I am %s %s\n", fname, lname)
}

main.go
package main

import "fmt"

func main() {
    var price int
    fmt.Println("Price?")
    fmt.Scanf("%d", &price)
    var rate float64
    fmt.Println("Rate?")
    fmt.Scanf("%f", &rate)
    fmt.Printf("Total: %f\n", float64(price) * rate)
}
